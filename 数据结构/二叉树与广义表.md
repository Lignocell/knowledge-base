# 广义表与二叉树

---

## 定义、广义表表示法

```
  ()        A或者A()        A(B,)或者A(B)      A(B,C)

 空树          A                  A              A
                                /              /  \
                               B              B    C
```



## 二、二叉树转广义表

```cpp
#define _CRT_SECURE_NO_WARNINGS
#include<iostream>
#define max 10
#include<time.h>//定义随机数
#include<string.h>
#define KEY(n)(n?n->key:-1)
//KEY(n)是宏定义，参数是n
//n是否为真，为真则传入key值，为假则为-1
using namespace std;


typedef struct node
{
    int key;
    struct node* node,*left, *right;
}node;


node* getnownode(int key)
{
    node* p = (node*)malloc(sizeof(node));
    p->key = key;
    p->left = p->right = NULL;
    return p;
}

node* insert(node* root, int key)
{
    if (root == NULL) return getnownode(key);
    if (rand() % 2) root->left = insert(root->left, key);
    else root->right  = insert(root->right, key);
    return root;
}
//插入随机个数进入二叉树
node* getrandomtree(int n)
{
    node* root = NULL;
    for (int i = 0; i < n; i++)
    {
        root = insert(root, rand() % 100);
    }
    return root;
}

void clear(node* root)
{
    if (root == NULL) return;
    clear(root->left);
    clear(root->right);
    free(root);
}



char buff[1000];
int len = 0;
//序列化
void serialize2(node* root)
{
    if (root == NULL) return;
    len += snprintf(buff + len, 100,"%d", root->key);
//buff指的是序列化后的存储空间，len指的是从buff数组中的len位置开始插入，100指的是输出的终点，root->key指需要插入的值，sprintf指返回值是1（因为插入了1个字符）len+sprintf()指更新len。

    if (root->left == NULL && root->right == NULL) return;
    len += snprintf(buff + len,100, "(");//输出左括号
    serialize2(root->left);
    if (root->right)//判断是否输出逗号
    {
        len += snprintf(buff + len,100, ",");
        serialize2(root->right);
   }
    //输出右括号
    len += snprintf(buff + len, 100,")");
}


void serialize(node* root)
{
    memset(buff, 0, sizeof(buff));
    //将指定位置设定为特定的值，buff指指向特定内存的指针，代表要设置的值，sizeof(buff)指设置需要的大小
    //memset的目的是去初始化buff
    len = 0;
    serialize2(root);
}


void print(node*root)
{
    printf("%d(%d,%d)\n", KEY(root), KEY(root->left), KEY(root->right));
}
//输出
void output(node* root)
{
    if (root == NULL) return;
    print( root);
    output(root->left);
    output(root->right);
}

int main()
{
    srand(time(0));
    node* root = getrandomtree(max);

    serialize(root);//序列化二叉树
    output(root);
    cout << "buff[]:"<<buff << endl;
    return 0;
}
```

## 三、广义表转二叉树

```cpp
//反序列化
node* deserialize(char* buff, int n)
{
    node** s = (node**)malloc(sizeof(node*) * max);//用数组代替一个栈
    int top = -1;//栈顶指针
    int flag = 0;//标记，flag==1==表明左子树，flag==2==表明右子树
    int scode = 0;//状态机
    node* p = NULL;//p永远指向最后一个生成的节点
    node* root = NULL;//root用来记录最后一个被弹出的节点，即整个二叉树的根节点
    for (int i = 0; buff[i]; i++)
    {
        switch (scode)
        {
        case 0://进行任务分发
        {
            if (buff[i] >= '0' && buff[i] <= '9') scode = 1;//1表明指向了节点
            else if (buff[i] == '(')scode = 2;
            else if (buff[i] == ',')scode = 3;
            else scode = 4;
            i--;//0不作为具体的buff【i】，进入case1时需要减一
        }
        break;
        case 1:
        {
            int key = 0;
            while (buff[i] >= '9' && buff[i] <= '0')
            {
                key = key * 10 + (buff[i] - '0');
                i++;
            }
            p = getnownode(key);
            if (top == 1 && flag == 0) s[top]->left = p;
            else if (top == 1 && flag == 1) s[top]->right = p;
            i--;//同理，要与循环抵消一个i
            scode = 0;
        }
        break;
        case 2://将元素压入栈中
        {
            s[top] = p;
            top++;
            flag = 0;
            scode = 0;
        }
        break;
        case 3://更新flag
        {
            flag = 1;
            scode = 0;
        }
        break;
        case 4://弹栈
        {
            root = s[top];
            top--;
            scode = 0;
        }
        break;
        }
    }
    return root;
}
```
